    <script>
        window.addEventListener('DOMContentLoaded', () => {
            const srcImage = new Image();
            srcImage.src = 'Rouss_Poleth_Makeup_Store.jpeg';
            
            srcImage.onload = () => {
                const W = srcImage.naturalWidth;
                const H = srcImage.naturalHeight;

                function extractShape(cropX, cropY, cropW, cropH, targetImgIds) {
                    const canvas = document.createElement('canvas');
                    canvas.width = cropW;
                    canvas.height = cropH;
                    const ctx = canvas.getContext('2d');

                    // Dibujar el fragmento recortado
                    ctx.drawImage(srcImage, cropX, cropY, cropW, cropH, 0, 0, cropW, cropH);

                    // Quitar el fondo beige para hacer transparente el dibujo
                    const imgData = ctx.getImageData(0, 0, cropW, cropH);
                    const data = imgData.data;

                    for (let i = 0; i < data.length; i += 4) {
                        const r = data[i];
                        const g = data[i + 1];
                        const b = data[i + 2];

                        // Si es el tono claro del fondo, lo volvemos transparente
                        if (r > 210 && g > 190 && b > 175) {
                            data[i + 3] = 0;
                        }
                    }

                    ctx.putImageData(imgData, 0, 0);
                    const dataUrl = canvas.toDataURL('image/png');

                    // Asignar a los elementos indicados
                    targetImgIds.forEach(id => {
                        const el = document.getElementById(id);
                        if (el) el.src = dataUrl;
                    });
                }

                // RECORTE ÚNICAMENTE DE LOS GRÁFICOS / DIBUJOS (EVITANDO LAS LETRAS DE ABAJO)
                // 1. Ojo Delineado (Esquina superior izquierda)
                extractShape(W * 0.18, H * 0.20, W * 0.28, H * 0.26, ['crop-eyeliner', 'hero-crop-1']);

                // 2. Labial y Beso (Esquina superior derecha)
                extractShape(W * 0.54, H * 0.20, W * 0.28, H * 0.26, ['crop-lipstick', 'hero-crop-2']);

                // 3. Pestaña y Brocha (Esquina inferior izquierda - solo gráfico)
                extractShape(W * 0.18, H * 0.48, W * 0.28, H * 0.25, ['crop-mascara', 'hero-crop-3']);

                // 4. Polvo y Rubor (Esquina inferior derecha - solo gráfico)
                extractShape(W * 0.54, H * 0.48, W * 0.28, H * 0.25, ['crop-powder', 'hero-crop-4']);
            };
        });
    </script>
